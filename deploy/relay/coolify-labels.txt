# Traefik labels for the relay on Coolify, behind Cloudflare (proxied DNS record).
# Add them to the resource's "Container Labels". Coolify already generates the router for the domain:
# replace <router> with its name (for example https-0-<uuid>) and keep any middlewares it already lists.
#
# Why both middlewares: the VPS IP is public, so a rate limit keyed on the client IP is only trustworthy if
# requests cannot skip Cloudflare. The allowlist checks the TCP peer (Cloudflare); then CF-Connecting-IP is
# set by Cloudflare and cannot be spoofed.

# 1) Only Cloudflare may reach the relay. Ranges from https://www.cloudflare.com/ips-v4 and
#    https://www.cloudflare.com/ips-v6 (checked 2026-11-10); update them when Cloudflare does.
traefik.http.middlewares.relay-cloudflare.ipallowlist.sourcerange=173.245.48.0/20,103.21.244.0/22,103.22.200.0/22,103.31.4.0/22,141.101.64.0/18,108.162.192.0/18,190.93.240.0/20,188.114.96.0/20,197.234.240.0/22,198.41.128.0/17,162.158.0.0/15,104.16.0.0/13,104.24.0.0/14,172.64.0.0/13,131.0.72.0/22,2400:cb00::/32,2606:4700::/32,2803:f800::/32,2405:b500::/32,2405:8100::/32,2a06:98c0::/29,2c0f:f248::/32

# 2) Attempt limit per real client IP: 30 requests per minute with a burst of 10. The owner reconnects about
#    once a minute per mailbox (plus a growing backoff after an error), far below the limit; a key-guessing
#    burst gets 429.
traefik.http.middlewares.relay-limit.ratelimit.average=30
traefik.http.middlewares.relay-limit.ratelimit.period=1m
traefik.http.middlewares.relay-limit.ratelimit.burst=10
traefik.http.middlewares.relay-limit.ratelimit.sourcecriterion.requestheadername=CF-Connecting-IP

# 3) Chain them on the HTTPS router Coolify generated (allowlist first).
traefik.http.routers.<router>.middlewares=relay-cloudflare,relay-limit
